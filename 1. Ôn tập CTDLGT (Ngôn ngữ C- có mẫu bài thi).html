Khai báo SNode, SList và các thao tác nền — C thuần (theo tr. 97–109 GT Thầy Thọ)

#include <stdio.h>
#include <stdlib.h>   /* malloc, free */
#include <math.h>     /* sqrt, pow — dùng khi cần kiểm tra số đặc biệt */
#include <string.h>

/* ---- Kiểu ItemType: có thể thay bằng float, struct SINHVIEN, v.v. ---- */
typedef int ItemType;

/* ---- Cấu trúc một nút trong DSLK đơn ---- */
typedef struct SNode {
    ItemType      Info;   /* dữ liệu của nút */
    struct SNode* Next;   /* con trỏ đến nút kế sau (NULL nếu cuối) */
} SNode;

/* ---- Cấu trúc quản lý toàn danh sách ---- */
typedef struct {
    SNode* Head;   /* địa chỉ nút đầu */
    SNode* Tail;   /* địa chỉ nút cuối */
} SList;

/* ========== CÁC HÀM NỀN ========== */

/* Tạo nút mới chứa giá trị x */
SNode* createSNode(ItemType x) {
    SNode* p = (SNode*)malloc(sizeof(SNode));
    if (p == NULL) { printf("Khong du bo nho!\n"); return NULL; }
    p->Info = x;
    p->Next = NULL;
    return p;
}

/* Khởi tạo danh sách rỗng */
void initSList(SList* sl) { sl->Head = sl->Tail = NULL; }

/* Kiểm tra danh sách rỗng: trả về 1 nếu rỗng, 0 nếu không */
int isEmpty(SList sl) { return (sl.Head == NULL) ? 1 : 0; }

/* Thêm nút p vào cuối danh sách */
int insertTail(SList* sl, SNode* p) {
    if (p == NULL) return 0;
    if (isEmpty(*sl)) { sl->Head = sl->Tail = p; }
    else { sl->Tail->Next = p; sl->Tail = p; }
    return 1;
}

/* In toàn bộ danh sách */
void showSList(SList sl) {
    SNode* p = sl.Head;
    while (p != NULL) { printf("%d ", p->Info); p = p->Next; }
    printf("\n");
}

/* Hủy toàn bộ danh sách, giải phóng bộ nhớ */
void deleteSList(SList* sl) {
    while (sl->Head != NULL) {
        SNode* p = sl->Head;
        sl->Head = sl->Head->Next;
        free(p);
    }
    sl->Tail = NULL;
}
⚠ Quy ước C thuần: malloc/free, không dùng new/delete Trong bài làm bằng C, dùng malloc(sizeof(SNode)) và free(p). Hàm nhận con trỏ đến SList (SList*) khi cần sửa, không dùng tham chiếu kiểu C++.

/* Kiểm tra số nguyên tố: trả về 1 nếu là số nguyên tố */
int isNguyenTo(int n) {
    int i;
    if (n < 2) return 0;
    for (i = 2; i * i <= n; i++)
        if (n % i == 0) return 0;
    return 1;
}

/* Kiểm tra số hoàn thiện: tổng các ước thực sự = n (VD: 6=1+2+3, 28=1+2+4+7+14) */
int isHoanThien(int n) {
    int i, tong = 1;
    if (n < 2) return 0;
    for (i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            tong += i;
            if (i != n / i) tong += n / i;
        }
    }
    return (tong == n);
}

/* Kiểm tra số chính phương: căn bậc 2 là số nguyên */
int isChinhPhuong(int n) {
    int s;
    if (n < 0) return 0;

    return (s * s == n);
}

/* Khi ItemType là struct — cách truy xuất p->Info.x */
typedef struct {
    float x, y;
} DIEM;

typedef DIEM ItemType;   /* thay typedef int ItemType; */

/* Khoảng cách Euclid giữa 2 điểm */
float khoangCach(DIEM a, DIEM b) {
    return (float)sqrt((a.x - b.x)*(a.x - b.x) + (a.y - b.y)*(a.y - b.y));
}

/* Tính chu vi đa giác (các đỉnh lưu theo thứ tự trong DSLK) */
float chuViDaGiac(SList sl) {
    float   cv = 0.0f;
    SNode* p  = sl.Head;
    if (p == NULL) return 0.0f;
    while (p->Next != NULL) {
        cv += khoangCach(p->Info, p->Next->Info);
        p = p->Next;
    }
    /* cạnh cuối: nối đỉnh cuối về đỉnh đầu */
    cv += khoangCach(sl.Tail->Info, sl.Head->Info);
    return cv;
}

    char  Ten[50];
    float DiemGK, DiemTL, DiemCK;

typedef SINHVIEN ItemType;

/* Tính điểm trung bình theo hệ số ví dụ: GK 30%, TL 20%, CK 50% */
float diemTB(SINHVIEN sv) {
    return sv.DiemGK * 0.3f + sv.DiemTL * 0.2f + sv.DiemCK * 0.5f;
}

/* Sắp xếp danh sách sinh viên giảm dần theo ĐTB bằng cách hoán vị Info */
void sapXepGiamTheoDTB(SList *sl) {
    SNode *p, *q;
    for (p = sl->Head; p != NULL; p = p->Next) {
        for (q = p->Next; q != NULL; q = q->Next) {
            if (diemTB(p->Info) < diemTB(q->Info)) {
                SINHVIEN temp = p->Info;
                p->Info = q->Info;
                q->Info = temp;
            }
        }
    }
}

/* In tối đa 3 sinh viên có ĐTB cao nhất sau khi đã sắp xếp */
void in3SVCaoNhat(SList sl) {
    int count = 0;
    SNode *p = sl.Head;
    while (p != NULL && count < 3) {
        printf("%s - %s - DTB: %.2f\n", p->Info.MaSV, p->Info.Ten, diemTB(p->Info));
        p = p->Next;
        count++;

}

/* Đảo ngược tại chỗ (thay đổi các con trỏ Next) */
void reverseList(SList* sl) {
    SNode* prev = NULL;
    SNode* curr = sl->Head;
    SNode* next = NULL;
    sl->Tail = sl->Head;       /* nút đầu trở thành nút cuối */
    while (curr != NULL) {
        next = curr->Next;     /* lưu nút kế */
        curr->Next = prev;     /* đảo chiều liên kết */
        prev = curr;
        curr = next;
    }
    sl->Head = prev;           /* nút cuối trở thành Head */
}

typedef struct {
    int tuSo;
    int mauSo;

int ucln(int a, int b) {
    if (a < 0) a = -a;
    if (b < 0) b = -b;
    while (b != 0) {
        int r = a % b;
        a = b;
        b = r;
    }
    return a == 0 ? 1 : a;
}

PHANSO rutGon(PHANSO p) {
    int d = ucln(p.tuSo, p.mauSo);
    p.tuSo /= d;
    p.mauSo /= d;
    if (p.mauSo < 0) { p.tuSo = -p.tuSo; p.mauSo = -p.mauSo; }
    return p;

PHANSO congPS(PHANSO a, PHANSO b) {
    PHANSO c;

Khai báo Stack bằng mảng — Ngôn ngữ C (theo tr. 141–143 GT Thầy Thọ)

#define MAXSIZE 100
typedef int ItemType;   /* cho Stack số nguyên */

typedef struct {
    ItemType Info[MAXSIZE];
    int      Top;   /* -1 khi rỗng */
} Stack;

void initStack(Stack* s) { s->Top = -1; }
int  isEmptyStack(Stack s) { return s.Top == -1; }
int  isFullStack(Stack s)  { return s.Top >= MAXSIZE - 1; }

int push(Stack* s, ItemType x) {
    if (isFullStack(*s)) return 0;
    s->Info[++(s->Top)] = x;
    return 1;
}

int pop(Stack* s, ItemType* x) {
    if (isEmptyStack(*s)) return 0;
    *x = s->Info[(s->Top)--];
    return 1;
}

int getTop(Stack s, ItemType* x) {
    if (isEmptyStack(s)) return 0;
    *x = s.Info[s.Top];
    return 1;
}
📌 Quy ước C thuần Hàm nhận con trỏ Stack* và ItemType* — không có tham chiếu &. Gọi hàm phải truyền địa chỉ: push(&s, x), pop(&s, &x).
Lý thuyết

Giải thuật Infix → Postfix bằng lời — CLO2 (4 ý × 0.25đ)

  1. Đọc tuần tự từng ký tự từ trái qua phải trong biểu thức Infix.
  2. Nếu ký tự là toán hạng (chữ số, biến): ghi ngay vào chuỗi kết quả (Postfix).
  3. Nếu là dấu mở ngoặc (: push vào Stack.
  4. Nếu là toán tử (+,−,×,÷): pop và ghi ra kết quả tất cả toán tử trên đỉnh Stack có độ ưu tiên ≥ toán tử hiện tại (trừ dấu (), rồi push toán tử hiện tại vào Stack.
    Ưu tiên: *, / = 2  >  +, - = 1  |  ( trong Stack = 0
  5. Nếu là dấu đóng ngoặc ): pop và ghi ra kết quả liên tục đến khi gặp (; pop ( ra nhưng không ghi vào kết quả.
  6. Hết biểu thức: pop tất cả phần còn lại trong Stack vào chuỗi kết quả.

#include <stdio.h>
#define MAXSIZE 100

typedef char ItemType;   /* Stack ký tự cho bài toán ngoặc */

typedef struct {
    ItemType Info[MAXSIZE];
    int      Top;
} Stack;

void initStack(Stack* s) { s->Top = -1; }
int  isEmptyStack(Stack s) { return s.Top == -1; }
void push(Stack* s, char c) { s->Info[++(s->Top)] = c; }
char pop(Stack* s) { return s->Info[(s->Top)--]; }
char topVal(Stack s) { return s.Info[s.Top]; }

/* Kiểm tra chuỗi ngoặc hợp lệ: trả về 1 nếu hợp lệ, 0 nếu không */
int kiemTraNgoac(const char* str) {
    Stack s;
    initStack(&s);
    int i;
    for (i = 0; str[i] != '\0'; i++) {
        char c = str[i];
        if (c == '(' || c == '[') {
            push(&s, c);
        } else if (c == ')') {
            if (isEmptyStack(s) || topVal(s) != '(') return 0;
            pop(&s);
        } else if (c == ']') {
            if (isEmptyStack(s) || topVal(s) != '[') return 0;
            pop(&s);
        }
    }
    return isEmptyStack(s);  /* rỗng = hợp lệ */
}

#define MAXQ 100
typedef int QItemType;

typedef struct {
    QItemType Info[MAXQ];
    int Front, Rear;
} Queue;

void initQueue(Queue *q) { q->Front = q->Rear = -1; }
int isEmptyQueue(Queue q) { return q.Front == -1; }
int isFullQueue(Queue q) { return (q.Rear + 1) % MAXQ == q.Front; }

int enQueue(Queue *q, QItemType x) {
    if (isFullQueue(*q)) return 0;
    if (isEmptyQueue(*q)) q->Front = q->Rear = 0;
    else q->Rear = (q->Rear + 1) % MAXQ;
    q->Info[q->Rear] = x;
    return 1;
}

int deQueue(Queue *q, QItemType *x) {
    if (isEmptyQueue(*q)) return 0;
    *x = q->Info[q->Front];
    if (q->Front == q->Rear) q->Front = q->Rear = -1;
    else q->Front = (q->Front + 1) % MAXQ;
    return 1;
}

/* Ví dụ hệ thống in: tiếp nhận 3 lệnh, hoàn thành 2 lệnh */
void moPhongMayIn() {
    Queue q;
    int job;
    initQueue(&q);
    enQueue(&q, 101); enQueue(&q, 102); enQueue(&q, 103);
    deQueue(&q, &job); printf("Da in lenh %d\n", job);
    deQueue(&q, &job); printf("Da in lenh %d\n", job);
}

4 Phép xoay AVL — Định nghĩa & Code C (tr. 181–186 GT Thầy Thọ)

LL — Xoay phải tại T

Mất CB lệch trái-trái: cây con phải của T1 (LH) gây ra.
→ Xoay phải T: T1 lên thay T.

Trước:  T [BF=-2]      Sau:  T1
       T1   R    →    L1   T
       L1  R1              R1 R

RR — Xoay trái tại T

Mất CB lệch phải-phải: cây con phải của T1 (RH) gây ra.
→ Xoay trái T: T1 lên thay T.

Trước: T [BF=+2]       Sau:  T1
      L    T1     →   T   R1
          L1 R1      L L1

LR — Xoay trái T1 rồi phải T

Mất CB lệch trái-phải: T lệch trái, nhưng T1 lại lệch phải.
→ Bước 1: rotateRR(T->Left). Bước 2: rotateLL(T).

RL — Xoay phải T1 rồi trái T

Mất CB lệch phải-trái: T lệch phải, nhưng T1 lại lệch trái.
→ Bước 1: rotateLL(T->Right). Bước 2: rotateRR(T).

/* ===== PHÉP XOAY LL (xoay phải tại T) — tr. 181 ===== */
void rotateLL(AVLNode** T) {
    AVLNode* T1 = (*T)->Left;
    (*T)->Left   = T1->Right;
    T1->Right    = *T;
    switch (T1->balFactor) {
        case LH: (*T)->balFactor = EH; T1->balFactor = EH; break;
        case EH: (*T)->balFactor = LH; T1->balFactor = RH; break;
    }
    *T = T1;
}

/* ===== PHÉP XOAY RR (xoay trái tại T) — tr. 186 ===== */
void rotateRR(AVLNode** T) {
    AVLNode* T1 = (*T)->Right;
    (*T)->Right  = T1->Left;
    T1->Left     = *T;
    switch (T1->balFactor) {
        case RH: (*T)->balFactor = EH; T1->balFactor = EH; break;
        case EH: (*T)->balFactor = RH; T1->balFactor = LH; break;
    }
    *T = T1;
}

/* ===== PHÉP XOAY LR (xoay kép: trái T1 rồi phải T) — tr. 182 ===== */
void rotateLR(AVLNode** T) {
    AVLNode* T1 = (*T)->Left;
    AVLNode* T2 = T1->Right;
    T1->Right    = T2->Left;
    T2->Left     = T1;
    (*T)->Left   = T2->Right;
    T2->Right    = *T;
    switch (T2->balFactor) {
        case LH: (*T)->balFactor = RH; T1->balFactor = EH; break;
        case EH: (*T)->balFactor = EH; T1->balFactor = EH; break;
        case RH: (*T)->balFactor = EH; T1->balFactor = LH; break;
    }
    T2->balFactor = EH;
    *T = T2;
}

/* ===== PHÉP XOAY RL (xoay kép: phải T1 rồi trái T) — tr. 184 ===== */
void rotateRL(AVLNode** T) {
    AVLNode* T1 = (*T)->Right;
    AVLNode* T2 = T1->Left;
    T1->Left     = T2->Right;
    T2->Right    = T1;
    (*T)->Right  = T2->Left;
    T2->Left     = *T;
    switch (T2->balFactor) {
        case RH: (*T)->balFactor = LH; T1->balFactor = EH; break;
        case EH: (*T)->balFactor = EH; T1->balFactor = EH; break;
        case LH: (*T)->balFactor = EH; T1->balFactor = RH; break;
    }
    T2->balFactor = EH;
    *T = T2;
}

/* Câu 1a.1: Đếm số phần tử chia hết cho 3 */
int demChiaHet3(SList sl) {
    int dem = 0;
    SNode *p = sl.Head;
    while (p != NULL) {
        if (p->Info % 3 == 0) dem++;
        p = p->Next;
    }
    return dem;
}

/* Câu 1a.2: Tính trung bình cộng các phần tử > 10 */
float trungBinhLonHon10(SList sl) {
    int dem = 0;
    long tong = 0;
    SNode *p = sl.Head;
    while (p != NULL) {
        if (p->Info > 10) {
            tong += p->Info;
            dem++;
        }
        p = p->Next;
    }
    return (dem == 0) ? 0.0f : (float)tong / dem;
}

/* Câu 1b: Tách danh sách thành danh sách chẵn và danh sách lẻ */
void tachChanLe(SList sl, SList *slChan, SList *slLe) {
    SNode *p = sl.Head;
    initSList(slChan);
    initSList(slLe);

    while (p != NULL) {
        SNode *q = createSNode(p->Info);   /* tạo node mới, không phá danh sách gốc */
        if (p->Info % 2 == 0)
            insertTail(slChan, q);
        else
            insertTail(slLe, q);
        p = p->Next;
    }
}

/* Quy ước: gốc ở mức k = 0 */
int demNutLaTaiMucK(TNode *root, int k) {
    if (root == NULL) return 0;

    if (k == 0) {
        if (root->Left == NULL && root->Right == NULL)
            return 1;
        return 0;

    return demNutLaTaiMucK(root->Left, k - 1)
         + demNutLaTaiMucK(root->Right, k - 1);

Câu 4

Bài giải mẫu — Bảng băm, Coalesced Hashing

4a. Lý thuyết ngắn gọn

Coalesced Hashing là phương pháp xử lý va chạm trong bảng băm bằng cách lưu phần tử ngay trong mảng băm và dùng thêm cột Next để liên kết các phần tử có cùng địa chỉ băm. Khi va chạm, tìm ô trống từ cuối bảng lên, đặt khóa vào ô trống đó và cập nhật Next của nút cuối chuỗi.

Ưu điểm: không cần cấp phát danh sách liên kết ngoài như Chaining; dữ liệu nằm trong một mảng nên tiết kiệm bộ nhớ ngoài và dễ mô phỏng trên giấy. Nhược điểm: thao tác xóa/cập nhật phức tạp hơn vì phải bảo toàn chuỗi Next.

Ghi chú học thuật — so sánh với CLRS / Sedgewick

  • Định nghĩa CSCB: GT Thầy Thọ: CSCB = h_phải − h_trái (RH − LH). CLRS và một số sách dùng ngược lại. Trong thi: theo đúng GT → LH = -1, EH = 0, RH = +1.
  • Mức cây (Level): trong code ôn tập này quy ước k=0 là gốc. Khi đề thi quy ước gốc ở mức 1 thì cần đổi điều kiện dừng tương ứng.
  • Trường hợp LB/RB trong AVL: Chỉ xảy ra khi xóa nút (không xảy ra khi thêm). GT Thầy Thọ gọi riêng là LB/RB; đây là điểm đặc thù của giáo trình, CLRS không phân biệt riêng nhưng logic tương đương.
  • Coalesced Hashing: GT Thầy Thọ tìm ô trống từ cuối bảng lên — đây là biến thể LISCH. Một số tài liệu tìm từ đầu xuống. Trong thi: theo GT = từ cuối lên.
  • Hàm băm tốt: tính toán nhanh, phân bố khóa tương đối đều và giảm xung đột. Khi làm bài, luôn tính đúng h(k), chỉ rõ vị trí va chạm và cách xử lý.
